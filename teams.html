function doGet(e) { const action = e && e.parameter ? e.parameter.action || "ping" : "ping"; if (action === "ping") { return ContentService .createTextOutput(JSON.stringify({ success: true, message: "KPL Season 03 API Connected" })) .setMimeType(ContentService.MimeType.JSON); } if (action === "teams") { const ss = SpreadsheetApp.openById( "1Sc3FNcCCTNezxmMiMuJamliAmFnVsLuHf71brYze-Rc" ); const sheet = ss.getSheetByName("Teams"); if (!sheet || sheet.getLastRow() < 2) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: "Teams sheet missing or empty" })) .setMimeType(ContentService.MimeType.JSON); } const values = sheet.getDataRange().getDisplayValues(); const headers = values.shift(); const teams = values .filter(row => row.some(cell => cell !== "")) .map(row => { const team = {}; headers.forEach((header, i) => { team[header] = row[i]; }); return team; }); return ContentService .createTextOutput(JSON.stringify({ success: true, teams: teams })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ success: false, message: "Unknown action: " + action })) .setMimeType(ContentService.MimeType.JSON); }
